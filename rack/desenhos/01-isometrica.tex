% isometric dimension: \cotaIso{x1}{y1}{x2}{y2}{dx}{dy}{text rotation}{text}
% points in sheet mm; (dx,dy) offsets the dimension line from the edge along an iso axis
\newcommand{\cotaIso}[8]{%
  \draw[linhaB] ({#1+0.15*(#5)},{#2+0.15*(#6)}) -- ({#1+1.2*(#5)},{#2+1.2*(#6)});
  \draw[linhaB] ({#3+0.15*(#5)},{#4+0.15*(#6)}) -- ({#3+1.2*(#5)},{#4+1.2*(#6)});
  \draw[setaDupla] ({#1+(#5)},{#2+(#6)}) -- ({#3+(#5)},{#4+(#6)});
  \node[txtcota, below, rotate=#7] at ({(#1+#3)/2+(#5)},{(#2+#4)/2+(#6)}) {#8};}

% visible faces of a box for a viewer at -X, -Y, +Z (X width, Y depth from the front, Z height):
% \faces[fill]{x1}{x2}{y1}{y2}{z1}{z2}, fill defaults to white
\newcommand{\faces}[7][white]{%
  \filldraw[linhaA, fill=#1] ({#2},{#4},{#6}) -- ({#2},{#5},{#6}) -- ({#2},{#5},{#7}) -- ({#2},{#4},{#7}) -- cycle;
  \filldraw[linhaA, fill=#1] ({#2},{#4},{#6}) -- ({#3},{#4},{#6}) -- ({#3},{#4},{#7}) -- ({#2},{#4},{#7}) -- cycle;
  \filldraw[linhaA, fill=#1] ({#2},{#4},{#7}) -- ({#3},{#4},{#7}) -- ({#3},{#5},{#7}) -- ({#2},{#5},{#7}) -- cycle;}

% upright: profile 5..584 with slots at 7..13 mm, connectors at 0..5 and 584..589: \upS{x1}{x2}{y1}{y2}
\newcommand{\upS}[4]{%
  \faces{#1}{#2}{#3}{#4}{0}{589}
  \fill[canal] ({#1},{#3+7},5) -- ({#1},{#3+13},5) -- ({#1},{#3+13},584) -- ({#1},{#3+7},584) -- cycle;
  \fill[canal] ({#1+7},{#3},5) -- ({#1+13},{#3},5) -- ({#1+13},{#3},584) -- ({#1+7},{#3},584) -- cycle;
  \foreach \o in {7,13}{
    \draw[linhaB] ({#1},{#3+\o},5) -- ({#1},{#3+\o},584);
    \draw[linhaB] ({#1+\o},{#3},5) -- ({#1+\o},{#3},584);}
  % top connector: left, front and top faces
  \faces[conector]{#1}{#2}{#3}{#4}{584}{589}
  % bottom connector: left and front strips only (its top face is hidden inside the upright)
  \filldraw[linhaA, fill=conector] ({#1},{#3},0) -- ({#1},{#4},0) -- ({#1},{#4},5) -- ({#1},{#3},5) -- cycle;
  \filldraw[linhaA, fill=conector] ({#1},{#3},0) -- ({#2},{#3},0) -- ({#2},{#3},5) -- ({#1},{#3},5) -- cycle;}
% width rail, long along X: slots on the front and top faces: \wrS{x1}{x2}{y1}{y2}{z1}{z2}
\newcommand{\wrS}[6]{%
  \faces{#1}{#2}{#3}{#4}{#5}{#6}
  \fill[canal] ({#1},{#3},{#5+7}) -- ({#2},{#3},{#5+7}) -- ({#2},{#3},{#5+13}) -- ({#1},{#3},{#5+13}) -- cycle;
  \fill[canal] ({#1},{#3+7},{#6}) -- ({#2},{#3+7},{#6}) -- ({#2},{#3+13},{#6}) -- ({#1},{#3+13},{#6}) -- cycle;
  \foreach \o in {7,13}{
    \draw[linhaB] ({#1},{#3},{#5+\o}) -- ({#2},{#3},{#5+\o});
    \draw[linhaB] ({#1},{#3+\o},{#6}) -- ({#2},{#3+\o},{#6});}}
% depth rail, long along Y: slots on the left and top faces: \drS{x1}{x2}{y1}{y2}{z1}{z2}
\newcommand{\drS}[6]{%
  \faces{#1}{#2}{#3}{#4}{#5}{#6}
  \fill[canal] ({#1},{#3},{#5+7}) -- ({#1},{#4},{#5+7}) -- ({#1},{#4},{#5+13}) -- ({#1},{#3},{#5+13}) -- cycle;
  \fill[canal] ({#1+7},{#3},{#6}) -- ({#1+13},{#3},{#6}) -- ({#1+13},{#4},{#6}) -- ({#1+7},{#4},{#6}) -- cycle;
  \foreach \o in {7,13}{
    \draw[linhaB] ({#1},{#3},{#5+\o}) -- ({#1},{#4},{#5+\o});
    \draw[linhaB] ({#1+\o},{#3},{#6}) -- ({#1+\o},{#4},{#6});}}

% L brackets of one side plane, across x1..x2: back vertices, back and front intermediate joints
% (no bracket at the front vertices): \ladoCant{x1}{x2}
\newcommand{\ladoCant}[2]{%
  \cantYZ{#1}{#2}{330}{569}{-\CantL}{-\CantL}
  \cantYZ{#1}{#2}{330}{20}{-\CantL}{\CantL}
  \foreach \z in {\EA,\EB}{
    \cantYZ{#1}{#2}{330}{\z+10}{-\CantL}{\CantL}
    \cantYZ{#1}{#2}{330}{\z-10}{-\CantL}{-\CantL}
    \cantYZ{#1}{#2}{20}{\z+10}{\CantL}{\CantL}
    \cantYZ{#1}{#2}{20}{\z-10}{\CantL}{-\CantL}}}
% L brackets of the back plane on one back upright, inner face at x, legs towards dx: \fundoCant{x}{dx}
\newcommand{\fundoCant}[2]{%
  \cantXZ{331}{349}{#1}{569}{#2}{-\CantL}
  \cantXZ{331}{349}{#1}{20}{#2}{\CantL}
  \foreach \z in {\EA,\EB}{
    \cantXZ{331}{349}{#1}{\z+10}{#2}{\CantL}
    \cantXZ{331}{349}{#1}{\z-10}{#2}{-\CantL}}}
